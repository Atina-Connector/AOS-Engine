function AOS_app()
% AOS_APP Lanzador modular de AOS Suite 0.1.9 R1.
% Puerta de entrada a la arquitectura 0.2.0 basada en bancos de trabajo.
  clc;
  iniciar_aos();
  AOS_root = fileparts(fileparts(fileparts(mfilename('fullpath'))));
  cd(AOS_root);

  while true
    imprimir_estado_local();
    fprintf('\n================================================================\n');
    fprintf(' AOS SUITE 0.1.9 R1 - PUERTA DE ENTRADA A AOS 0.2.0\n');
    fprintf(' AESIR Oilfield Systems | Plataforma modular GNU Octave\n');
    fprintf('================================================================\n');
    fprintf(' 1 - AOS SLA          %s\n', aos_suite_etiqueta_producto('SLA'));
    fprintf(' 2 - AOS WELLS        %s\n', aos_suite_etiqueta_producto('WELLS'));
    fprintf(' 3 - AOS CAD          %s\n', aos_suite_etiqueta_producto('CAD'));
    fprintf(' 4 - AOS NETWORKS     %s\n', aos_suite_etiqueta_producto('NETWORKS'));
    fprintf(' 5 - AOS ELECTRICAL   %s\n', aos_suite_etiqueta_producto('ELECTRICAL'));
    fprintf(' 6 - AOS FACILITIES   %s\n', aos_suite_etiqueta_producto('FACILITIES'));
    fprintf(' 7 - AOS GEOLOGY      %s\n', aos_suite_etiqueta_producto('GEOLOGY'));
    fprintf(' 8 - AOS FLUIDS       %s\n', aos_suite_etiqueta_producto('FLUIDS'));
    fprintf(' 9 - AOS SCADA        %s\n', aos_suite_etiqueta_producto('SCADA'));
    fprintf('10 - AOS MAINTENANCE  %s\n', aos_suite_etiqueta_producto('MAINTENANCE'));
    fprintf('11 - AOS DATA         %s\n', aos_suite_etiqueta_producto('DATA'));
    fprintf('12 - AOS SOLVERS      %s\n', aos_suite_etiqueta_producto('SOLVERS'));
    fprintf('13 - AOS GLOBAL       %s\n', aos_suite_etiqueta_producto('GLOBAL'));
    fprintf('14 - ROADMAP GENERAL DE AOS\n');
    fprintf('15 - Configuracion, versiones y diagnosticos\n');
    fprintf('16 - AOS VIEWER       %s\n', aos_suite_etiqueta_producto('VIEWER'));
    fprintf(' 0 - Salir\n');

    op = aos_leer_opcion('Seleccione [0-16]: ', []);
    switch op
      case 1, AOS_menu_SLA();
      case 2, AOS_menu_wells();
      case 3, AOS_menu_cad_topologia();
      case 4, AOS_menu_networks();
      case 5, AOS_menu_electrical();
      case 6, AOS_menu_facilities();
      case 7, AOS_menu_geology();
      case 8, AOS_menu_fluidos();
      case 9, AOS_menu_scada();
      case 10, AOS_menu_maintenance();
      case 11, AOS_menu_data();
      case 12, AOS_menu_solvers();
      case 13, AOS_menu_global();
      case 14, AOS_menu_roadmap();
      case 15, AOS_menu_suite_config_diag();
      case 16, AOS_menu_viewer();
      case 0
        if confirmar_salida_local(), break; endif
      otherwise
        fprintf('Opcion no valida.\n');
    endswitch
  endwhile
endfunction

function imprimir_estado_local()
  global CONFIG_ACTIVA AOSDAT_ACTIVO geologia;
  global ULTIMO_QL ULTIMO_QO ULTIMO_QINY ULTIMO_TIPO ULTIMO_PARAM;
  if ~isempty(CONFIG_ACTIVA) && isstruct(CONFIG_ACTIVA)
    try
      CONFIG_ACTIVA = aos_normalizar_config(CONFIG_ACTIVA, 'GENERAL');
      aos_sincronizar_geologia_activa();
    catch err
      fprintf('ADVERTENCIA AOS: %s\n', err.message);
    end_try_catch
    try
      aos_menu_imprimir_resumen_config(CONFIG_ACTIVA, AOSDAT_ACTIVO, geologia);
    catch
    end_try_catch
  endif
  if ~isempty(ULTIMO_QL) && isstruct(ULTIMO_PARAM)
    try
      aos_menu_imprimir_ultima_corrida(ULTIMO_PARAM, ULTIMO_QL, ULTIMO_QO, ...
        ULTIMO_QINY, ULTIMO_TIPO);
    catch
    end_try_catch
  endif
endfunction

function salir = confirmar_salida_local()
  global CONFIG_ACTIVA AOSDAT_ACTIVO geologia;
  global ULTIMO_QL ULTIMO_QO ULTIMO_QINY ULTIMO_TIPO ULTIMO_PARAM;
  salir = false;
  r = lower(strtrim(input('Salir de AOS Suite? (s/n) [n]: ', 's')));
  if ~strcmp(r, 's'), return; endif
  limpiar = lower(strtrim(input('Eliminar la configuracion activa de memoria? (s/n) [n]: ', 's')));
  if strcmp(limpiar, 's')
    CONFIG_ACTIVA = []; AOSDAT_ACTIVO = ''; geologia = [];
    ULTIMO_QL = []; ULTIMO_QO = []; ULTIMO_QINY = [];
    ULTIMO_TIPO = []; ULTIMO_PARAM = [];
  endif
  fprintf('Saliendo de AOS Suite...\n');
  salir = true;
endfunction
